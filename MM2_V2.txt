local function track(connection)
    table.insert(Connections, connection)
    return connection
end

local function disconnectAll()
    for _, connection in ipairs(Connections) do
        pcall(function() connection:Disconnect() end)
    end
    table.clear(Connections)
end

local function tween(instance, duration, properties)
    local info = TweenInfo.new(duration or 0.18, Enum.EasingStyle.Quart, Enum.EasingDirection.Out)
    local anim = TweenService:Create(instance, info, properties)
    anim:Play()
    return anim
end

local function create(className, properties, parent)
    local object = Instance.new(className)
    for key, value in pairs(properties or {}) do
        object[key] = value
    end
    object.Parent = parent
    return object
end

local function round(parent, radius)
    return create("UICorner", { CornerRadius = UDim.new(0, radius or 10) }, parent)
end

local function outline(parent, color, transparency)
    return create("UIStroke", { Color = color or Color3.fromRGB(57, 62, 78), Transparency = transparency or 0, Thickness = 1 }, parent)
end

local function getCharacter() return LocalPlayer.Character end
local function getHumanoid()
    local char = getCharacter()
    return char and char:FindFirstChildOfClass("Humanoid")
end
local function getRoot(char)
    char = char or getCharacter()
    return char and char:FindFirstChild("HumanoidRootPart")
end

local function addToggle(pageName, title, description, initial, onChanged)
    local card = create("Frame", { Size = UDim2.new(1, -6, 0, 65), BackgroundColor3 = Theme.Card, BorderSizePixel = 0 }, PageFrames[pageName])
    round(card, 10)
    create("TextLabel", { Position = UDim2.fromOffset(12, 8), Size = UDim2.new(1, -125, 0, 22), BackgroundTransparency = 1, Text = title, TextColor3 = Theme.Text, TextSize = 12, Font = Enum.Font.GothamSemibold, TextXAlignment = Enum.TextXAlignment.Left }, card)
    create("TextLabel", { Position = UDim2.fromOffset(12, 31), Size = UDim2.new(1, -125, 0, 22), BackgroundTransparency = 1, Text = description or "", TextColor3 = Theme.Muted, TextSize = 10, Font = Enum.Font.Gotham, TextXAlignment = Enum.TextXAlignment.Left, TextWrapped = true }, card)
    
    local action = create("TextButton", { AnchorPoint = Vector2.new(1, 0.5), Position = UDim2.new(1, -10, 0.5, 0), Size = UDim2.fromOffset(94, 32), BackgroundColor3 = initial and Theme.Accent or Theme.Hover, Text = initial and "ON" or "OFF", TextColor3 = Theme.Text, TextSize = 10, Font = Enum.Font.GothamBold, AutoButtonColor = false }, card)
    round(action, 8)

    local val = initial or false
    track(action.Activated:Connect(function()
        val = not val
        action.Text = val and "ON" or "OFF"
        action.BackgroundColor3 = val and Theme.Accent or Theme.Hover
        if onChanged then onChanged(val) end
    end))

    Controls[title] = { Set = function(v) val = v action.Text = v and "ON" or "OFF" action.BackgroundColor3 = v and Theme.Accent or Theme.Hover end }
    table.insert(FeatureRegistry, { Title = title, Description = description or "", Card = card })
end

local function addSlider(pageName, title, description, minV, maxV, initial, onChanged)
    local card = create("Frame", { Size = UDim2.new(1, -6, 0, 86), BackgroundColor3 = Theme.Card, BorderSizePixel = 0 }, PageFrames[pageName])
    round(card, 10)
    create("TextLabel", { Position = UDim2.fromOffset(12, 8), Size = UDim2.new(1, -100, 0, 20), BackgroundTransparency = 1, Text = title, TextColor3 = Theme.Text, TextSize = 12, Font = Enum.Font.GothamSemibold, TextXAlignment = Enum.TextXAlignment.Left }, card)
    create("TextLabel", { Position = UDim2.fromOffset(12, 29), Size = UDim2.new(1, -24, 0, 18), BackgroundTransparency = 1, Text = description or "", TextColor3 = Theme.Muted, TextSize = 10, Font = Enum.Font.Gotham, TextXAlignment = Enum.TextXAlignment.Left }, card)
    
    local valLabel = create("TextLabel", { AnchorPoint = Vector2.new(1, 0), Position = UDim2.new(1, -12, 0, 8), Size = UDim2.fromOffset(64, 20), BackgroundTransparency = 1, Text = tostring(initial), TextColor3 = Theme.Accent, TextSize = 12, Font = Enum.Font.GothamBold, TextXAlignment = Enum.TextXAlignment.Right }, card)
    local bar = create("Frame", { Position = UDim2.new(0, 12, 1, -20), Size = UDim2.new(1, -24, 0, 6), BackgroundColor3 = Theme.Hover, BorderSizePixel = 0 }, card)
    round(bar, 4)

    local ratio = (initial - minV) / (maxV - minV)
    local fill = create("Frame", { Size = UDim2.new(ratio, 0, 1, 0), BackgroundColor3 = Theme.Accent, BorderSizePixel = 0 }, bar)
    round(fill, 4)

    local function updateVal(v)
        v = math.clamp(v, minV, maxV)
        valLabel.Text = tostring(v)
        fill.Size = UDim2.new((v - minV) / (maxV - minV), 0, 1, 0)
        if onChanged then onChanged(v) end
    end

    local dragging = false
    track(bar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then dragging = true end
    end))
    track(UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local pct = math.clamp((input.Position.X - bar.AbsolutePosition.X) / math.max(bar.AbsoluteSize.X, 1), 0, 1)
            updateVal(math.floor(minV + pct * (maxV - minV) + 0.5))
        end
    end))
    track(UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then dragging = false end
    end))

    table.insert(FeatureRegistry, { Title = title, Description = description or "", Card = card })
end

--// Logic & Features Implementation
local function getCollectibles()
    local res = {}
    for _, inst in ipairs(CollectionService:GetTagged(CONFIG.CollectibleTag)) do
        if inst and inst:IsDescendantOf(Workspace) then
            local p = inst:IsA("BasePart") and inst or inst:FindFirstChildWhichIsA("BasePart", true)
            if p then table.insert(res, { Instance = inst, Part = p }) end
        end
    end
    return res
end

local function teleportToGun()
    local root = getRoot()
    local gun = Workspace:FindFirstChild(CONFIG.DroppedGunName, true)
    if root and gun then
        local p = gun:IsA("BasePart") and gun or gun:FindFirstChildWhichIsA("BasePart", true)
        if p then root.CFrame = p.CFrame + CONFIG.TeleportOffset notify("Teleported to dropped gun!", Theme.Sheriff) return end
    end
    notify("Gun not found on map", Theme.Red)
end

-- Home Actions
addFeature("Home", "Teleport to Nearest Item", "Teleports to the closest collectible/coin.", function()
    local root = getRoot()
    local items = getCollectibles()
    if root and #items > 0 then
        root.CFrame = items[1].Part.CFrame + CONFIG.TeleportOffset
        notify("Teleported to item")
    else
        notify("No items found")
    end
end)

-- Visuals Tab
addToggle("Visuals", "Role ESP Highlights", "Highlights players based on their roles.", false, function(on)
    State.RoleESP = on
    if on then
        track(RunService.Heartbeat:Connect(function()
            if not State.RoleESP then return end
            for _, plr in ipairs(Players:GetPlayers()) do
                if plr ~= LocalPlayer and plr.Character then
                    local h = RoleHighlights[plr]
                    if not h then
                        h = Instance.new("Highlight")
                        h.FillTransparency = 0.5
                        h.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                        h.Parent = plr.Character
                        RoleHighlights[plr] = h
                    end
                    h.FillColor = Theme.Green -- Default Innocent color placeholder
                end
            end
        end))
    else
        for _, h in pairs(RoleHighlights) do if h then h:Destroy() end end
        table.clear(RoleHighlights)
    end
end)

    local q = string.lower(UI.Search.Text)
    for _, item in ipairs(FeatureRegistry) do

-- UI Toggles

    State.UIVisible = not State.UIVisible
    UI.Main.Visible = State.UIVisible
end

    if not processed and input.KeyCode == CONFIG.ToggleKey then toggleUI() end
end))
