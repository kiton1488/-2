local function create(class, props, parent)
    local inst = Instance.new(class)
    for k, v in pairs(props) do inst[k] = v end
    inst.Parent = parent
    return inst
end

local function round(inst, radius)
    create("UICorner", { CornerRadius = UDim.new(0, radius) }, inst)
end

local function stroke(inst, color, thickness)
    return create("UIStroke", { Color = color or Theme.Container, Thickness = thickness or 1, ApplyStrokeMode = Enum.ApplyStrokeMode.Border }, inst)
end

local conns = {}
local function connect(name, signal, fn)
    if conns[name] then conns[name]:Disconnect() end
    conns[name] = signal:Connect(fn)
end
local function disconnect(name)
    if conns[name] then conns[name]:Disconnect() conns[name] = nil end
end
local function startLoop(name, interval, fn)
    if conns[name] then conns[name]:Disconnect() end
    local acc = 0
    conns[name] = RunService.Heartbeat:Connect(function(dt)
        acc += dt
        if acc >= interval then acc = 0 fn() end
    end)
end

local function getRoot(plr)

    return char and char:FindFirstChild("HumanoidRootPart")
end
local function getHumanoid(plr)
    local char = plr.Character
    return char and char:FindFirstChildOfClass("Humanoid")
end
local function nameHas(name, list)
    name = name:lower()
    for _, kw in ipairs(list) do if name:find(kw, 1, true) then return true end end
    return false
end

local function hasTool(plr, name)
    local char = plr.Character
    if char and char:FindFirstChild(name) then return true end
    if plr == player then
        local bp = plr:FindFirstChildOfClass("Backpack")
        return bp ~= nil and bp:FindFirstChild(name) ~= nil
    end
    return false
end
local function knownRole(plr)
    if Config.RoleAttribute then
        local a = plr:GetAttribute(Config.RoleAttribute)
        if a then return a end
    end
    if hasTool(plr, Config.Items.Knife) then return Config.Roles.Murderer end
    if hasTool(plr, Config.Items.Gun) then return Config.Roles.Sheriff end
    return nil
end
local function getRole(plr) return knownRole(plr) or Config.Roles.Innocent end
local function isMurderer(plr) return getRole(plr) == Config.Roles.Murderer end
local function roleColor(role)
    if role == Config.Roles.Murderer then return Colors.Murderer end
    if role == Config.Roles.Sheriff then return Colors.Sheriff end
    return Colors.Innocent
end

local function isPickable(inst)
    if inst:IsA("BasePart") and not inst.CanTouch then return false end
    if inst:GetAttribute("Collectible") == true then return true end
    for _, d in ipairs(inst:GetDescendants()) do
        if d:IsA("TouchTransmitter") or d:IsA("ClickDetector") or d:IsA("ProximityPrompt") or d:IsA("LuaSourceContainer") then
            return true
        end
    end
    return false
end
local function isCollectible(inst)
    local cfg = Config.Collectibles
    if CollectionService:HasTag(inst, cfg.Tag) then return true end
    if cfg.TagOnly then return false end
    if not (inst:IsA("BasePart") or inst:IsA("Model")) then return false end
    local name = inst.Name
    if not nameHas(name, cfg.Keywords) or nameHas(name, cfg.Exclude) or nameHas(name, cfg.BlacklistKeywords) then return false end
    local anc = inst.Parent
    while anc and anc ~= Workspace do
        if nameHas(anc.Name, cfg.BlacklistKeywords) then return false end
        if anc:IsA("Model") then
            if Players:GetPlayerFromCharacter(anc) or nameHas(anc.Name, cfg.Keywords) then return false end
        end
        anc = anc.Parent
    end
    if cfg.RequireInteraction and not isPickable(inst) then return false end
    return true
end

local tracked, visited = {}, {}
local trackerUsers, trackerCount, scanId = {}, 0, 0

local function classify(inst)
    if inst.Name == Config.Items.DroppedGun and (inst:IsA("BasePart") or inst:IsA("Model")) then return "gun" end
    if isCollectible(inst) then return "item" end
    return nil
end
local function track(inst)
    local k = classify(inst)
    if k then tracked[inst] = k end
end
local function untrack(inst) tracked[inst] = nil visited[inst] = nil end

local function startTracker(user)
    if trackerUsers[user] then return end
    trackerUsers[user] = true
    trackerCount += 1
    if trackerCount > 1 then return end
    scanId += 1
    local id = scanId
    local tag = Config.Collectibles.Tag
    connect("TrackAdded", Workspace.DescendantAdded, track)
    connect("TrackRemoved", Workspace.DescendantRemoving, untrack)
    connect("TagAdded", CollectionService:GetInstanceAddedSignal(tag), function(i)
        if i:IsDescendantOf(Workspace) then tracked[i] = "item" end
    end)
    connect("TagRemoved", CollectionService:GetInstanceRemovedSignal(tag), untrack)
    for _, i in ipairs(CollectionService:GetTagged(tag)) do
        if i:IsDescendantOf(Workspace) then tracked[i] = "item" end
    end
    task.spawn(function()
        local n = 0
        for _, d in ipairs(Workspace:GetDescendants()) do
            if id ~= scanId then return end
            track(d)
            n += 1
            if n % 400 == 0 then task.wait() end
        end
    end)
end
local function stopTracker(user)
    if not trackerUsers[user] then return end
    trackerUsers[user] = nil
    trackerCount -= 1
    if trackerCount == 0 then
        scanId += 1
        disconnect("TrackAdded") disconnect("TrackRemoved")
        disconnect("TagAdded") disconnect("TagRemoved")
        table.clear(tracked) table.clear(visited)
    end
end

-- DRAGGABLE
-- ============================================================
local function makeDraggable(handle, target)
    local state = { Dragged = false }
    local activeInput, dragStart, startPos
    local function point(inp)
        if inp.UserInputType == Enum.UserInputType.Touch then
            return Vector2.new(inp.Position.X, inp.Position.Y)
        end
        return UserInputService:GetMouseLocation()
    end
    handle.InputBegan:Connect(function(input)
        local t = input.UserInputType
        if t == Enum.UserInputType.MouseButton1 or t == Enum.UserInputType.Touch then
            activeInput = input
            state.Dragged = false
            dragStart = point(input)
            startPos = target.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End and activeInput == input then
                    activeInput = nil
                end
            end)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if not activeInput then return end
        local current
        if activeInput.UserInputType == Enum.UserInputType.Touch then
            if input ~= activeInput then return end
            current = point(input)
        elseif input.UserInputType == Enum.UserInputType.MouseMovement then
            current = UserInputService:GetMouseLocation()
        else
            return
        end
        local delta = current - dragStart
        if delta.Magnitude > 6 then state.Dragged = true end
        target.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end)
    return state
end

local tabs = {}
local currentTab

local function createTab(id, title)
    local btn = create("TextButton", {
        BackgroundColor3 = Theme.Sidebar,
        Size = UDim2.new(1, 0, 0, 32),
        Font = Enum.Font.GothamMedium,
        Text = "  " .. title,
        TextColor3 = Theme.SubText,
        TextSize = 12,
        TextXAlignment = Enum.TextXAlignment.Left,
        AutoButtonColor = false,
    }, tabScroll)
    round(btn, 6)

    local page = create("Frame", {
        BackgroundTransparency = 1,
        Size = UDim2.new(1, -8, 0, 0),
        AutomaticSize = Enum.AutomaticSize.Y,
        Visible = false,
    }, pagesHolder)
    create("UIListLayout", { Padding = UDim.new(0, 6), SortOrder = Enum.SortOrder.LayoutOrder }, page)
    create("UIPadding", { PaddingLeft = UDim.new(0, 8), PaddingRight = UDim.new(0, 8) }, page)

    local tab = { id = id, title = title, btn = btn, page = page, order = #tabs + 1 }
    btn.LayoutOrder = tab.order
    page.LayoutOrder = tab.order

    btn.MouseEnter:Connect(function()
        if currentTab ~= tab then btn.BackgroundColor3 = Theme.Container end
    end)
    btn.MouseLeave:Connect(function()
        if currentTab ~= tab then btn.BackgroundColor3 = Theme.Sidebar end
    end)
    btn.MouseButton1Click:Connect(function()
        for _, t in pairs(tabs) do
            if t == tab then
                t.btn.BackgroundColor3 = Theme.Accent
                t.btn.TextColor3 = Theme.Text
                t.page.Visible = true
                pageTitle.Text = t.title
                currentTab = t
            else
                t.btn.BackgroundColor3 = Theme.Sidebar
                t.btn.TextColor3 = Theme.SubText
                t.page.Visible = false
            end
        end
    end)

    table.insert(tabs, tab)
    return page, tab
end

local allToggles = {}
local function addToggle(page, text, default, callback)
    local card = makeCard(page, 40)
    local label = create("TextLabel", {
        BackgroundTransparency = 1,
        Position = UDim2.fromOffset(12, 0),
        Size = UDim2.new(1, -70, 1, 0),
        Font = Enum.Font.Gotham,
        Text = text,
        TextColor3 = Theme.Text,
        TextSize = 13,
        TextXAlignment = Enum.TextXAlignment.Left,
    }, card)

    local track = create("Frame", {
        AnchorPoint = Vector2.new(1, 0.5),
        BackgroundColor3 = Theme.Background,
        Position = UDim2.new(1, -12, 0.5, 0),
        Size = UDim2.fromOffset(40, 20),
        ClipsDescendants = true,
    }, card)
    round(track, 10)

    local knob = create("Frame", {
        BackgroundColor3 = Theme.SubText,
        Position = UDim2.fromOffset(2, 2),
        Size = UDim2.fromOffset(16, 16),
    }, track)
    round(knob, 8)

    local clickArea = create("TextButton", {
        BackgroundTransparency = 1,
        Size = UDim2.fromScale(1, 1),
        Text = "",
    }, card)

    local state = default or false
    local function refresh(anim)
        local targetTrack = state and Theme.Accent or Theme.Background
        local targetKnob = state and Theme.Text or Theme.SubText
        local targetPos = state and UDim2.fromOffset(22, 2) or UDim2.fromOffset(2, 2)
        if anim then
            TweenService:Create(track, TweenInfo.new(0.15), { BackgroundColor3 = targetTrack }):Play()
            TweenService:Create(knob, TweenInfo.new(0.15), { BackgroundColor3 = targetKnob, Position = targetPos }):Play()
        else
            track.BackgroundColor3 = targetTrack
            knob.BackgroundColor3 = targetKnob
            knob.Position = targetPos
        end
    end

    local function set(value)
        state = value
        refresh(true)
        callback(state)
    end

    clickArea.MouseButton1Click:Connect(function() set(not state) end)
    refresh(false)
    table.insert(allToggles, { setter = set, refresh = refresh })

    return set
end

-- Slider
local function addSlider(page, text, minV, maxV, default, step, callback, formatter)
    local card = makeCard(page, 56)
    local label = create("TextLabel", {
        BackgroundTransparency = 1,
        Position = UDim2.fromOffset(12, 6),
        Size = UDim2.new(1, -90, 0, 18),
        Font = Enum.Font.Gotham,
        Text = text,
        TextColor3 = Theme.Text,
        TextSize = 12,
        TextXAlignment = Enum.TextXAlignment.Left,
    }, card)

    local valueLbl = create("TextLabel", {
        BackgroundTransparency = 1,
        Position = UDim2.new(1, -80, 0, 6),
        Size = UDim2.fromOffset(68, 18),
        Font = Enum.Font.GothamBold,
        Text = tostring(default),
        TextColor3 = Theme.Accent,
        TextSize = 12,
        TextXAlignment = Enum.TextXAlignment.Right,
    }, card)

    local bar = create("Frame", {
        BackgroundColor3 = Theme.Background,
        Position = UDim2.new(0, 12, 1, -20),
        Size = UDim2.new(1, -24, 0, 6),
    }, card)
    round(bar, 3)

    local fill = create("Frame", {
        BackgroundColor3 = Theme.Accent,
        Size = UDim2.fromScale((default - minV) / (maxV - minV), 1),
    }, bar)
    round(fill, 3)

    local knob = create("Frame", {
        AnchorPoint = Vector2.new(0.5, 0.5),
        BackgroundColor3 = Theme.Text,
        Position = UDim2.fromScale((default - minV) / (maxV - minV), 0.5),
        Size = UDim2.fromOffset(14, 14),
        ZIndex = 2,
    }, bar)
    round(knob, 7)

    local value = default
    local function fmt(v)
        if formatter then return formatter(v) end
        return tostring(math.floor(v * 100 + 0.5) / 100)
    end
    local function setValue(v, fireCallback)
        v = math.clamp(v, minV, maxV)
        if step and step > 0 then v = math.floor((v - minV) / step + 0.5) * step + minV end
        value = v
        local t = (v - minV) / (maxV - minV)
        fill.Size = UDim2.fromScale(t, 1)
        knob.Position = UDim2.fromScale(t, 0.5)
        valueLbl.Text = fmt(v)
        if fireCallback then callback(v) end
    end

    local dragging = false
    local function updateFromInput(input)
        local absX = input.Position.X
        local barPos = bar.AbsolutePosition.X
        local barW = bar.AbsoluteSize.X
        local t = math.clamp((absX - barPos) / barW, 0, 1)
        setValue(minV + t * (maxV - minV), true)
    end

    bar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            updateFromInput(input)
        end
    end)
    bar.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            updateFromInput(input)
        end
    end)

    setValue(default, false)
    return setValue
end

local function updateItemEsp()
    local root = getRoot(player)
    table.clear(sortedItems)
    if root then
        local origin = root.Position
        for inst, kind in pairs(tracked) do
            if inst.Parent then
                local pos = inst:GetPivot().Position
                sortedItems[#sortedItems + 1] = { inst = inst, kind = kind, dist = (pos - origin).Magnitude }
            else
                untrack(inst)
            end
        end
        table.sort(sortedItems, function(a, b)
            if a.kind ~= b.kind then return a.kind == "gun" end
            return a.dist < b.dist
        end)
    end
    local shown, gunSeen = 0, false
    for _, entry in ipairs(sortedItems) do
        if shown >= Config.ItemESP.MaxLabels then break end
        local adornee = getAdornee(entry.inst)
        if adornee then
            shown += 1
            local isGun = entry.kind == "gun"
            local color = isGun and Colors.Sheriff or Colors.Items
            local bb = getLabel(shown)
            bb.Adornee = adornee
            bb.Enabled = true
            bb.Label.Text = ("%s [%d]"):format(isGun and "GUN" or entry.inst.Name, math.floor(entry.dist))
            bb.Label.TextColor3 = color
            if isGun then
                gunSeen = true
                if not gunHighlight then
                    gunHighlight = create("Highlight", { FillTransparency = 0.4, DepthMode = Enum.HighlightDepthMode.AlwaysOnTop })
                end
                gunHighlight.FillColor = color
                if gunHighlight.Parent ~= entry.inst then gunHighlight.Parent = entry.inst end
            end
        end
    end
    for i = shown + 1, #labelPool do labelPool[i].Enabled = false end
    if gunHighlight and not gunSeen then gunHighlight.Parent = nil end
end
local function setItemEsp(on)
    State.itemEspEnabled = on
    if on then
        itemFolder = create("Folder", { Name = "HubItemESP" }, playerGui)
        startTracker("ItemESP")
        startLoop("ItemESP", Config.ItemESP.Interval, updateItemEsp)
    else
        disconnect("ItemESP")
        stopTracker("ItemESP")
        table.clear(labelPool) table.clear(sortedItems)
        if gunHighlight then gunHighlight:Destroy() gunHighlight = nil end
        if itemFolder then itemFolder:Destroy() itemFolder = nil end
    end
end

-- Hitbox
local hitboxOriginal = {}
local function restoreHitbox(part)
    local o = hitboxOriginal[part]
    if o then
        if part.Parent then
            part.Size, part.Transparency, part.CanCollide = o.Size, o.Transparency, o.CanCollide
        end
        hitboxOriginal[part] = nil
    end
end
local function applyHitboxes()
    local size = Vector3.new(State.hitboxSize, State.hitboxSize, State.hitboxSize)
    for part in pairs(hitboxOriginal) do
        if not part:IsDescendantOf(Workspace) then hitboxOriginal[part] = nil end
    end
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= player then
            local root = getRoot(plr)
            if root then
                if State.hitboxMode == "All" or isMurderer(plr) then
                    if not hitboxOriginal[root] then
                        hitboxOriginal[root] = { Size = root.Size, Transparency = root.Transparency, CanCollide = root.CanCollide }
                    end
                    root.Size = size
                    root.Transparency = Config.Hitbox.Transparency
                    root.CanCollide = false
                else
                    restoreHitbox(root)
                end
            end
        end
    end
end
local function setHitbox(on)
    State.hitboxEnabled = on
    if on then
        applyHitboxes()
        startLoop("Hitbox", 0.5, applyHitboxes)
    else
        disconnect("Hitbox")
        for part in pairs(hitboxOriginal) do restoreHitbox(part) end
    end
end

local rayParams = RaycastParams.new()
rayParams.FilterType = Enum.RaycastFilterType.Exclude

local function findTarget(cam, myChar, cfg, filter, byDistance)
    local center = cam.ViewportSize / 2
    local origin = cam.CFrame.Position
    rayParams.FilterDescendantsInstances = { myChar }
    local bestPart, bestScore = nil, math.huge
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= player and filter(plr) then
            local tChar = plr.Character
            local hum = tChar and tChar:FindFirstChildOfClass("Humanoid")
            local part = tChar and (tChar:FindFirstChild("Head") or tChar:FindFirstChild("HumanoidRootPart"))
            if part and hum and hum.Health > 0 then
                local screen, onScreen = cam:WorldToViewportPoint(part.Position)
                local offset = part.Position - origin
                if onScreen and offset.Magnitude <= cfg.Range then
                    local pixelDist = (Vector2.new(screen.X, screen.Y) - center).Magnitude
                    if pixelDist <= cfg.FovPixels then
                        local score = byDistance and offset.Magnitude or pixelDist
                        if score < bestScore then
                            local hit = Workspace:Raycast(origin, offset, rayParams)
                            if not hit or hit.Instance:IsDescendantOf(tChar) then
                                bestPart, bestScore = part, score
                            end
                        end
                    end
                end
            end
        end
    end
    return bestPart
end

    if on and not aimBound[bindName] then
        aimBound[bindName] = true
        RunService:BindToRenderStep(bindName, Enum.RenderPriority.Camera.Value + 1, step)
    elseif not on and aimBound[bindName] then

    end
end

local function shootTarget()
    local char = player.Character
    if not char then return end
    local gun = char:FindFirstChild(Config.Items.Gun)
    if not gun or not gun:IsA("Tool") then
        notify("No gun equipped", Theme.Red)
        return
    end
    local cam = Workspace.CurrentCamera
    local target = findTarget(cam, char, Config.AutoShoot, isMurderer, false)
    if not target then return end
    cam.CFrame = CFrame.lookAt(cam.CFrame.Position, target.Position)
    gun:Activate()
end

local lastShotAt = 0
local function shootStep()
    local char = player.Character
    if not char then return end
    local gun = char:FindFirstChild(Config.Items.Gun)
    if not gun or not gun:IsA("Tool") then return end
    local cam = Workspace.CurrentCamera
    local target = findTarget(cam, char, Config.AutoShoot, isMurderer, false)
    if not target then return end
    local aimPoint = target.Position + target.AssemblyLinearVelocity * Config.AutoShoot.LeadTime
    cam.CFrame = cam.CFrame:Lerp(CFrame.lookAt(cam.CFrame.Position, aimPoint), Config.AutoShoot.Smoothness)
    local align = cam.CFrame.LookVector:Dot((aimPoint - cam.CFrame.Position).Unit)
    local now = os.clock()
    if align >= Config.AutoShoot.MinAlignment and now - lastShotAt >= Config.AutoShoot.Cooldown then
        lastShotAt = now
        if State.shotType == "Silent Aim" then
            cam.CFrame = CFrame.lookAt(cam.CFrame.Position, aimPoint)
        end
        gun:Activate()
    end
end

local function throwStep()
    local char = player.Character
    if not char then return end
    local knife = char:FindFirstChild(Config.Items.Knife)
    if not knife or not knife:IsA("Tool") then return end
    local cam = Workspace.CurrentCamera
    local target = findTarget(cam, char, Config.AutoThrow, function() return true end, true)
    if not target then return end
    local aimPoint = target.Position + target.AssemblyLinearVelocity * Config.AutoThrow.LeadTime
    cam.CFrame = cam.CFrame:Lerp(CFrame.lookAt(cam.CFrame.Position, aimPoint), Config.AutoThrow.Smoothness)
    local align = cam.CFrame.LookVector:Dot((aimPoint - cam.CFrame.Position).Unit)
    local now = os.clock()
    if align >= Config.AutoThrow.MinAlignment and now - lastShotAt >= Config.AutoThrow.Cooldown then
        lastShotAt = now
        knife:Activate()
    end
end

-- Dodge
local dodgeReadyAt = 0
local function isSwinging(char)
    local knife = char:FindFirstChild(Config.Items.Knife)
    if not knife then return false end
    if knife:GetAttribute(Config.Dodge.SwingAttribute) then return true end
    local hum = char:FindFirstChildOfClass("Humanoid")
    local anim = hum and hum:FindFirstChildOfClass("Animator")
    if anim then
        for _, track in ipairs(anim:GetPlayingAnimationTracks()) do
            if nameHas(track.Name, Config.Dodge.AnimKeywords) then return true end
        end
    end
    return false
end
local function dodgeStep()
    local root = getRoot(player)
    if not root or os.clock() < dodgeReadyAt then return end
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= player and isMurderer(plr) then
            local tRoot = getRoot(plr)
            if tRoot and (tRoot.Position - root.Position).Magnitude <= Config.Dodge.Radius and isSwinging(plr.Character) then
                local away = (root.Position - tRoot.Position) * Vector3.new(1, 0, 1)
                if away.Magnitude < 0.1 then away = -root.CFrame.LookVector end
                local dir = away.Unit
                local dist = Config.Dodge.Distance
                rayParams.FilterDescendantsInstances = { player.Character }
                local hit = Workspace:Raycast(root.Position, dir * dist, rayParams)
                if hit then dist = math.max(hit.Distance - 2, 0) end
                if dist > 1 then
                    root.CFrame = root.CFrame + dir * dist
                    dodgeReadyAt = os.clock() + Config.Dodge.Cooldown
                    notify("Dodged!", Theme.Green)
                end
                return
            end
        end
    end
end
local function setDodge(on)
    State.dodgeEnabled = on
    if on then startLoop("Dodge", Config.Dodge.Interval, dodgeStep)
    else disconnect("Dodge") end
end

local function killAll()
    if State.killAllRunning then notify("Kill All already running", Theme.Red) return end
    if not equipKnife() then
        notify("Knife not found in inventory", Theme.Red)
        return
    end
    State.killAllRunning = true
    notify("Kill All started", Theme.Red)
    task.spawn(function()
        while State.killAllRunning do
            local root = getRoot(player)
            if not root then break end
            local knife = player.Character and player.Character:FindFirstChild(Config.Items.Knife)
            if not knife then
                if not equipKnife() then break end
                knife = player.Character and player.Character:FindFirstChild(Config.Items.Knife)
            end
            local target, tRoot, bestDist = nil, nil, math.huge
            for _, plr in ipairs(Players:GetPlayers()) do
                if plr ~= player then
                    local r = getRoot(plr)
                    local hum = getHumanoid(plr)
                    if r and hum and hum.Health > 0 then
                        local d = (r.Position - root.Position).Magnitude
                        if d < bestDist then
                            target, tRoot, bestDist = plr, r, d
                        end
                    end
                end
            end
            if not target or not tRoot then
                notify("Kill All complete", Theme.Green)
                break
            end
            root.CFrame = tRoot.CFrame * CFrame.new(0, 0, 2.5) -- позади цели
            task.wait(0.05)
            local cam = Workspace.CurrentCamera
            cam.CFrame = CFrame.lookAt(cam.CFrame.Position, tRoot.Position)
            if knife then knife:Activate() end
            task.wait(State.killAllDelay)
        end
        State.killAllRunning = false
        notify("Kill All stopped", Theme.SubText)
    end)
end
local function stopKillAll() State.killAllRunning = false end

local flinging = false
local function flingPlayer(target)
    if flinging or not target then return end
    local root, tRoot = getRoot(player), getRoot(target)
    if not root or not tRoot then notify("Target unavailable", Theme.Red) return end
    flinging = true
    local home = root.CFrame
    local startT = os.clock()
    connect("Fling", RunService.Heartbeat, function()
        local r, tr = getRoot(player), target and getRoot(target)
        if not r or not tr or os.clock() - startT >= Config.Fling.Duration then
            disconnect("Fling")
            if r then
                r.AssemblyAngularVelocity = Vector3.zero
                r.AssemblyLinearVelocity = Vector3.zero
                r.CFrame = home
            end
            flinging = false
            return
        end
        r.CFrame = tr.CFrame

-- Fly
local flyVel, flyGyro
local flyUpUntil = 0
local function destroyFly()
    if flyVel then flyVel:Destroy() flyVel = nil end
    if flyGyro then flyGyro:Destroy() flyGyro = nil end
    local hum = getHumanoid(player)
    if hum then hum.PlatformStand = false end
end
local function flyStep()
    local root, hum, cam = getRoot(player), getHumanoid(player), Workspace.CurrentCamera
    if not (root and hum and cam) then return end
    if not flyVel or flyVel.Parent ~= root then
        destroyFly()
        flyVel = create("BodyVelocity", { MaxForce = Vector3.one * 1e5, Velocity = Vector3.zero }, root)
        flyGyro = create("BodyGyro", { MaxTorque = Vector3.one * 1e5, P = 1e4, CFrame = cam.CFrame }, root)
    end
    hum.PlatformStand = true
    local cf = cam.CFrame
    local dir = Vector3.zero
    if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir += cf.LookVector end
    if UserInputService:IsKeyDown(Enum.KeyCode.S) then dir -= cf.LookVector end
    if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir += cf.RightVector end
    if UserInputService:IsKeyDown(Enum.KeyCode.A) then dir -= cf.RightVector end
    if UserInputService:IsKeyDown(Enum.KeyCode.Space) or os.clock() < flyUpUntil then dir += Vector3.yAxis end
    if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then dir -= Vector3.yAxis end
    local move = hum.MoveDirection
    if dir.Magnitude == 0 and move.Magnitude > 0 then
        local flatLook = Vector3.new(cf.LookVector.X, 0, cf.LookVector.Z)
        if flatLook.Magnitude > 0 then
            flatLook = flatLook.Unit
            local flatRight = Vector3.new(cf.RightVector.X, 0, cf.RightVector.Z).Unit
            dir = cf.LookVector * move:Dot(flatLook) + cf.RightVector * move:Dot(flatRight)
        end
    end
    flyVel.Velocity = dir.Magnitude > 0 and dir.Unit * State.flySpeed or Vector3.zero
    flyGyro.CFrame = cf
end
local function setFly(on)
    State.flyEnabled = on
    if on then
        connect("FlyJump", UserInputService.JumpRequest, function()
            flyUpUntil = os.clock() + 0.25
        end)
        connect("Fly", RunService.RenderStepped, flyStep)
    else
        disconnect("Fly") disconnect("FlyJump")
        destroyFly()
    end
end

local function farmStep()
    local root, hum = getRoot(player), getHumanoid(player)
    if not root or not hum or hum.Health <= 0 then return end
    local now = os.clock()
    local origin = root.Position
    local nearest, nearestPos, best = nil, nil, math.huge
    for inst, kind in pairs(tracked) do
        if kind == "item" then
            if not inst.Parent then
                untrack(inst)
            elseif not visited[inst] or now - visited[inst] > Config.Farm.RevisitSeconds then
                local pos = inst:GetPivot().Position
                local d = (pos - origin).Magnitude
                if d < best then nearest, nearestPos, best = inst, pos, d end
            end
        end
    end
    if nearest then
        visited[nearest] = now
        local off = Vector3.new(
            (math.random() - 0.5) * 2 * Config.Farm.SideOffset,
            Config.Farm.HeightOffset,
            (math.random() - 0.5) * 2 * Config.Farm.SideOffset
        )
        local rot = root.CFrame - root.CFrame.Position
        root.CFrame = CFrame.new(nearestPos + off) * rot
    end
end
local function setFarm(on)
    State.farmEnabled = on
    if on then
        startTracker("Farm")
        startLoop("Farm", State.farmDelay, farmStep)
    else
        disconnect("Farm")
        stopTracker("Farm")
    end
end

local function saveConfig(name)
    name = name ~= "" and name or "default"
    local data = {
        espEnabled = State.espEnabled,
        tracerEnabled = State.tracerEnabled,
        itemEspEnabled = State.itemEspEnabled,
        walkSpeed = State.walkSpeed,
        jumpPower = State.jumpPower,
        flySpeed = State.flySpeed,
        hitboxSize = State.hitboxSize,
        killAllDelay = State.killAllDelay,
        farmDelay = State.farmDelay,
        showMobileButtons = State.showMobileButtons,
        binds = {
            ShootMurderer = State.binds.ShootMurderer.Name,
            KillAll = State.binds.KillAll.Name,
            GrabGun = State.binds.GrabGun.Name,
            ToggleUI = State.binds.ToggleUI.Name,
        },
    }
    local ok, enc = pcall(HttpService.JSONEncode, HttpService, data)
    if ok then
        pcall(function()
            if not isfolder or not isfolder("MM2Hub") then
                if makefolder then makefolder("MM2Hub") end
            end
            if writefile then writefile("MM2Hub/" .. name .. ".json", enc) end
        end)
        notify("Config saved: " .. name, Theme.Green)
    else
        notify("Save failed", Theme.Red)
    end
end

local function loadConfig(name)
    name = name ~= "" and name or "default"
    local ok, content = pcall(function() return readfile and readfile("MM2Hub/" .. name .. ".json") end)
    if not ok or not content then notify("Config not found", Theme.Red) return end
    local ok2, data = pcall(HttpService.JSONDecode, HttpService, content)
    if not ok2 or type(data) ~= "table" then notify("Load failed", Theme.Red) return end
    if data.walkSpeed then State.walkSpeed = data.walkSpeed end
    if data.jumpPower then State.jumpPower = data.jumpPower end
    if data.flySpeed then State.flySpeed = data.flySpeed end
    if data.hitboxSize then State.hitboxSize = data.hitboxSize end
    if data.killAllDelay then State.killAllDelay = data.killAllDelay end
    if data.farmDelay then State.farmDelay = data.farmDelay end
    if data.binds then
        for k, v in pairs(data.binds) do
            local ok3, key = pcall(function() return Enum.KeyCode[v] end)
            if ok3 and key then State.binds[k] = key end
        end
    end
    notify("Config loaded: " .. name, Theme.Green)
end

        Position = UDim2.fromOffset(12, 4),
        Size = UDim2.new(1, -60, 0, 20),
        Font = Enum.Font.GothamMedium,
        Text = title,
        TextColor3 = Theme.Text,
        TextSize = 12,
        TextXAlignment = Enum.TextXAlignment.Left,
    }, row)
    local swatch = create("Frame", {
        Position = UDim2.new(1, -40, 0, 6),
        Size = UDim2.fromOffset(28, 16),
        BackgroundColor3 = Colors[key],
    }, row)
    round(swatch, 4)
    local labels = {}
    local channels = { "R", "G", "B" }
    local function refresh()
        local c = Colors[key]
        swatch.BackgroundColor3 = c
        labels.R.Text = "R " .. math.round(c.R * 255)
        labels.G.Text = "G " .. math.round(c.G * 255)
        labels.B.Text = "B " .. math.round(c.B * 255)
    end
    local function bump(ch, delta)
        local c = Colors[key]
        local r, g, b = c.R * 255, c.G * 255, c.B * 255
        if ch == "R" then r = math.clamp(r + delta, 0, 255)
        elseif ch == "G" then g = math.clamp(g + delta, 0, 255)
        else b = math.clamp(b + delta, 0, 255) end
        Colors[key] = Color3.fromRGB(math.round(r), math.round(g), math.round(b))
        refresh()
        onColorsChanged()
    end
    for i, ch in ipairs(channels) do
        local cell = create("Frame", {
            BackgroundTransparency = 1,
            Position = UDim2.new((i - 1) / 3, 4, 0, 28),
            Size = UDim2.new(1 / 3, -8, 0, 24),
        }, row)
        labels[ch] = create("TextLabel", {
            BackgroundTransparency = 1,
            Size = UDim2.new(1, -40, 1, 0),
            Font = Enum.Font.Gotham,
            TextColor3 = Theme.SubText,
            TextSize = 11,
            TextXAlignment = Enum.TextXAlignment.Left,
        }, cell)
        for _, spec in ipairs({ { "-", -38, -15 }, { "+", -2, 15 } }) do
            local b = create("TextButton", {
                AnchorPoint = Vector2.new(1, 0.5),
                Position = UDim2.new(1, spec[2], 0.5, 0),
                Size = UDim2.fromOffset(17, 20),
                BackgroundColor3 = Theme.Background,
                Font = Enum.Font.GothamBold,
                Text = spec[1],
                TextColor3 = Theme.Text,
                TextSize = 12,
            }, cell)
            round(b, 4)
            b.MouseButton1Click:Connect(function() bump(ch, spec[3]) end)
        end
    end
    refresh()
    table.insert(colorRefreshers, refresh)
end

-- ============================================================
-- SEARCH FILTER
-- ============================================================
searchInput:GetPropertyChangedSignal("Text"):Connect(function()
    local q = searchInput.Text:lower()
    for _, tab in ipairs(tabs) do
        local match = q == "" or tab.title:lower():find(q, 1, true) ~= nil
        if match then
            tab.btn.Visible = true
        else
            -- Also check inner buttons/labels
            local innerMatch = false
            for _, desc in ipairs(tab.page:GetDescendants()) do
                if desc:IsA("TextLabel") or desc:IsA("TextButton") then
                    if desc.Text and desc.Text ~= "" and desc.Text:lower():find(q, 1, true) then
                        innerMatch = true
                        break
                    end
                end
            end
            tab.btn.Visible = innerMatch
        end
    end
end)
